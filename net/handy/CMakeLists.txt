# mpnet-handy: UDP ping reflector + HTTP counters in one epoll loop, on handy (fork github.com/JebetrexxMC/handy)
cmake_minimum_required(VERSION 3.14)
project(mpnet_handy CXX)
set(CMAKE_CXX_STANDARD 11)
include(FetchContent)
FetchContent_Declare(handy GIT_REPOSITORY https://github.com/JebetrexxMC/handy.git GIT_TAG master GIT_SHALLOW TRUE)
FetchContent_GetProperties(handy)
if(NOT handy_POPULATED)
  FetchContent_Populate(handy)
endif()
file(GLOB HANDY_SRCS ${handy_SOURCE_DIR}/handy/*.cc)
list(FILTER HANDY_SRCS EXCLUDE REGEX ".*_test\\.cc$")
add_library(handy_s STATIC ${HANDY_SRCS})
target_include_directories(handy_s PUBLIC ${handy_SOURCE_DIR})
if(APPLE)
  target_compile_definitions(handy_s PUBLIC OS_MACOSX)
else()
  target_compile_definitions(handy_s PUBLIC OS_LINUX)
endif()
find_package(Threads REQUIRED)
target_link_libraries(handy_s PUBLIC Threads::Threads)
add_executable(mpnet-handy mpnet-handy.cc)
target_link_libraries(mpnet-handy handy_s)
